local BASE = "https://raw.githubusercontent.com/moshixzn/priv/main/FolderName/"

-- Override require to fetch from GitHub
local _require = require
local cache = {}
function require(path)
    path = path:gsub("%.", "/")
    if cache[path] then return cache[path] end

    -- try .lua.txt first then .lua
    local url = BASE .. path .. ".lua.txt"
    local ok, result = pcall(function()
        return loadstring(game:HttpGet(url))()
    end)
    if not ok then
        local ok2, result2 = pcall(function()
            return loadstring(game:HttpGet(BASE .. path .. ".lua"))()
        end)
        if ok2 then result = result2 end
    end

    cache[path] = result
    return result
end

-- fix shared for executors that dont have it (Delta etc)
if not shared then shared = {} end

-- Initialize Luraph globals if they do not exist
pcall(function()
    loadstring("getfenv().LPH_NO_VIRTUALIZE = function(...) return ... end")()
end)

getfenv().PP_SCRAMBLE_NUM = function(...) return ... end
getfenv().PP_SCRAMBLE_STR = function(...) return ... end
getfenv().PP_SCRAMBLE_RE_NUM = function(...) return ... end

local ok1, Profiler = pcall(require, "Utility/Profiler")
if not ok1 then
    warn("Failed to load Profiler: " .. tostring(Profiler))
    Profiler = { run = function(_, fn, ...) return fn(...) end }
end

local ok2, Lycoris = pcall(require, "Lycoris")
if not ok2 then
    warn("Failed to load Lycoris: " .. tostring(Lycoris))
    return
end

local function initializeScript()
    if shared.Lycoris then
        shared.Lycoris.detach()
        Lycoris.queued = shared.Lycoris.queued
    end
    shared.Lycoris = Lycoris
    shared.Lycoris.init()
end

local function onInitializeError(err)
    warn("Failed to initialize.")
    warn(err)
    warn(debug.traceback())
    pcall(function() Lycoris.detach() end)
end

Profiler.run("Main_InitializeScript", function(...)
    return xpcall(initializeScript, onInitializeError, ...)
end)
