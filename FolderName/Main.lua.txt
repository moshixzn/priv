local BASE = "https://raw.githubusercontent.com/moshixzn/priv/main/"

-- Override require to fetch from GitHub
local _require = require
local cache = {}
function require(path)
    path = path:gsub("%.", "/")
    if cache[path] then return cache[path] end
    local url = BASE .. path .. ".lua.txt"
    local ok, result = pcall(function()
        return loadstring(game:HttpGet(url))()
    end)
    if not ok then
        result = loadstring(game:HttpGet(BASE .. path .. ".lua"))()
    end
    cache[path] = result
    return result
end

-- No shared table = nothing to do
if not shared then
    return warn("No shared, no script.")
end

-- Initialize Luraph + preprocessor globals into the TRUE global env
-- so every module loaded via require() can see them immediately.
-- Using getgenv() instead of loadstring+getfenv() fixes the environment
-- isolation bug where LPH_NO_VIRTUALIZE was invisible to submodules.
getgenv().LPH_NO_VIRTUALIZE   = function(...) return ... end
getgenv().PP_SCRAMBLE_NUM      = function(...) return ... end
getgenv().PP_SCRAMBLE_STR      = function(...) return ... end
getgenv().PP_SCRAMBLE_RE_NUM   = function(...) return ... end

---@module Utility.Profiler
local Profiler = require("Utility/Profiler")

---@module Lycoris
local Lycoris = require("Lycoris")

local function initializeScript()
    if shared.Lycoris then
        shared.Lycoris.detach()
        Lycoris.queued = shared.Lycoris.queued
    end
    shared.Lycoris = Lycoris
    shared.Lycoris.init()
end

---@param error string
local function onInitializeError(error)
    warn("Failed to initialize.")
    warn(error)
    warn(debug.traceback())
    Lycoris.detach()
end

Profiler.run("Main_InitializeScript", function(...)
    return xpcall(initializeScript, onInitializeError, ...)
end)
